TypeScript Configuration
Keep the TypeScript implementation independently buildable.

package.json

{
  "type": "module",
  "scripts": {
    "build": "tsc",
    "start": "npm run build && node --enable-source-maps dist/server.js"
  },
  "devDependencies": {
    "@types/node": "^26.4.0",
    "typescript": "^7.0.2"
  }
}

tsconfig.json

{
  "compilerOptions": {
    "rootDir": "./src",
    "outDir": "./dist",
    "target": "ES2022",
    "module": "ES2022",
    "lib": ["ES2022", "DOM"],
    "types": ["node"],
    "strict": true,
    "sourceMap": true
  },
  "include": ["src/**/*.ts"]
}
