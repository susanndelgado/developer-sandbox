Starter Configuration
Keep the project independently buildable.

package.json

{
  "type": "module",
  "scripts": { "build": "tsc" },
  "devDependencies": { "typescript": "^7.0.2" }
}

tsconfig.json

{
  "compilerOptions": {
    "rootDir": "./src",
    "outDir": "./dist",
    "target": "ES2022",
    "module": "ES2022",
    "lib": ["ES2022", "DOM"],
    "strict": true,
    "sourceMap": true
  },
  "include": ["src/**/*.ts"]
}
